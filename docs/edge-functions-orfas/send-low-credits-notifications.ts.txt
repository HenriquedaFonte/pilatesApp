Deno.serve(async (req)=>{
  // Handle CORS preflight requests
  if (req.method === 'OPTIONS') {
    return new Response('ok', {
      headers: corsHeaders
    });
  }
  try {
    // Initialize Supabase client
    const supabaseClient = createClient(Deno.env.get('SUPABASE_URL') ?? '', Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? '');
    // Get request parameters
    const { notification_type = 'low_credits', test_mode = false } = await req.json().catch(()=>({}));
    console.log(`Starting email notification process for type: ${notification_type}`);
    // Get students eligible for notifications
    const { data: students, error: studentsError } = await supabaseClient.rpc('get_students_for_notification', {
      notification_type
    });
    if (studentsError) {
      console.error('Error fetching students:', studentsError);
      throw studentsError;
    }
    if (!students || students.length === 0) {
      console.log('No students found for notification');
      return new Response(JSON.stringify({
        success: true,
        message: 'No students require notifications at this time',
        students_processed: 0
      }), {
        headers: {
          ...corsHeaders,
          'Content-Type': 'application/json'
        },
        status: 200
      });
    }
    console.log(`Found ${students.length} students eligible for notifications`);
    // Email service configuration
    const emailServiceUrl = Deno.env.get('EMAIL_SERVICE_URL') || 'https://api.resend.com/emails';
    const emailApiKey = Deno.env.get('EMAIL_API_KEY') || Deno.env.get('RESEND_API_KEY');
    const studioName = Deno.env.get('STUDIO_NAME') || 'Studio de Pilates';
    const studioEmail = Deno.env.get('STUDIO_EMAIL') || 'contato@studiopilates.com';
    const studioPhone = Deno.env.get('STUDIO_PHONE') || '(11) 99999-9999';
    const purchaseLink = Deno.env.get('PURCHASE_LINK') || 'https://studiopilates.com/creditos';
    if (!emailApiKey) {
      throw new Error('Email API key not configured');
    }
    const results = {
      success: 0,
      failed: 0,
      errors: []
    };
    // Process each student
    for (const student of students){
      try {
        console.log(`Processing student: ${student.student_name} (${student.student_email}) - Language: ${student.preferred_language}`);
        // Generate email template based on notification type and language
        const emailTemplate = generateEmailTemplate(student, notification_type, student.preferred_language || 'pt', studioName, studioPhone, studioEmail, purchaseLink);
        // Prepare email data
        const emailData = {
          from: `${studioName} <${studioEmail}>`,
          to: [
            student.student_email
          ],
          subject: emailTemplate.subject,
          html: emailTemplate.html,
          text: emailTemplate.text
        };
        let emailResult;
        if (test_mode) {
          // In test mode, just log what would be sent
          console.log('TEST MODE - Would send email:', {
            to: student.student_email,
            subject: emailTemplate.subject,
            language: student.preferred_language,
            credits: {
              individual: student.individual_credits,
              duo: student.duo_credits,
              group: student.group_credits,
              total: student.total_credits
            }
          });
          emailResult = {
            id: `test-${Date.now()}`
          };
        } else {
          // Send actual email
          const emailResponse = await fetch(emailServiceUrl, {
            method: 'POST',
            headers: {
              'Authorization': `Bearer ${emailApiKey}`,
              'Content-Type': 'application/json'
            },
            body: JSON.stringify(emailData)
          });
          if (!emailResponse.ok) {
            const errorText = await emailResponse.text();
            throw new Error(`Email service error: ${emailResponse.status} - ${errorText}`);
          }
          emailResult = await emailResponse.json();
        }
        // Log successful email
        const { error: logError } = await supabaseClient.rpc('log_email_notification', {
          p_student_id: student.student_id,
          p_email_type: notification_type,
          p_recipient_email: student.student_email,
          p_subject: emailTemplate.subject,
          p_status: 'sent',
          p_error_message: null,
          p_credits_at_time: JSON.stringify({
            individual: student.individual_credits,
            duo: student.duo_credits,
            group: student.group_credits,
            total: student.total_credits,
            threshold: student.low_credits_threshold,
            language: student.preferred_language
          }),
          p_email_provider_id: emailResult.id
        });
        if (logError) {
          console.error('Error logging email:', logError);
        }
        results.success++;
        console.log(`Successfully sent email to ${student.student_name} in ${student.preferred_language}`);
      } catch (error) {
        console.error(`Failed to send email to ${student.student_name}:`, error);
        // Log failed email
        await supabaseClient.rpc('log_email_notification', {
          p_student_id: student.student_id,
          p_email_type: notification_type,
          p_recipient_email: student.student_email,
          p_subject: `Failed: ${notification_type} notification`,
          p_status: 'failed',
          p_error_message: error.message,
          p_credits_at_time: JSON.stringify({
            individual: student.individual_credits,
            duo: student.duo_credits,
            group: student.group_credits,
            total: student.total_credits,
            threshold: student.low_credits_threshold,
            language: student.preferred_language
          }),
          p_email_provider_id: null
        });
        results.failed++;
        results.errors.push(`${student.student_name}: ${error.message}`);
      }
    }
    console.log('Email notification process completed:', results);
    return new Response(JSON.stringify({
      success: true,
      message: `Processed ${students.length} students`,
      results: {
        students_processed: students.length,
        emails_sent: results.success,
        emails_failed: results.failed,
        errors: results.errors
      }
    }), {
      headers: {
        ...corsHeaders,
        'Content-Type': 'application/json'
      },
      status: 200
    });
  } catch (error) {
    console.error('Error in email notification function:', error);
    return new Response(JSON.stringify({
      success: false,
      error: error.message
    }), {
      headers: {
        ...corsHeaders,
        'Content-Type': 'application/json'
      },
      status: 500
    });
  }
});
function generateEmailTemplate(student, notificationType, language, studioName, studioPhone, studioEmail, purchaseLink) {
  const isZeroCredits = student.total_credits === 0;
  const lang = translations[language] || translations['pt'] // Fallback to Portuguese
  ;
  if (isZeroCredits || notificationType === 'zero_credits') {
    return {
      subject: lang.zeroCreditsSubject.replace('{studioName}', studioName),
      html: generateZeroCreditsHtml(student, language, studioName, studioPhone, studioEmail, purchaseLink),
      text: generateZeroCreditsText(student, language, studioName, studioPhone, studioEmail, purchaseLink)
    };
  } else {
    return {
      subject: lang.lowCreditsSubject.replace('{studioName}', studioName),
      html: generateLowCreditsHtml(student, language, studioName, studioPhone, studioEmail, purchaseLink),
      text: generateLowCreditsText(student, language, studioName, studioPhone, studioEmail, purchaseLink)
    };
  }
}
